\section*{Acronyms}
\addcontentsline{toc}{section}{Acronyms}
\begin{description}
  \item[CI/CD] Continuous Integration / Continuous Delivery
  \item[DM] Diebold--Mariano (test of equal predictive accuracy)
  \item[DNN] Deep Neural Network
  \item[ENTSO-E] European Network of Transmission System Operators for Electricity
  \item[EPF] Electricity Price Forecasting
  \item[GW] Giacomini--White (test of conditional predictive ability)
  \item[LEAR] Lasso Estimated AutoRegressive (model)
  \item[MAE] Mean Absolute Error
  \item[MLOps] Machine Learning Operations
  \item[rMAE] relative Mean Absolute Error
  \item[RMSE] Root Mean Squared Error
  \item[sMAPE] symmetric Mean Absolute Percentage Error
  \item[TPE] Tree-structured Parzen Estimator
  \item[VST] Variance-Stabilising Transformation
\end{description}
